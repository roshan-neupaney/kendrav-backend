                List posts (with filters — status, date range)

    message: string,

                id: number;
                media_url: string;
                media_type: string;

                post_id: number;
                created_id: string;
                updated_at: string;
            }];
            channel_posts: [{

                post_id: number;

                published_at: string;

                error_message: string;
                created_at: string;
                updated_at: string;
            }]
            is_active: true;
            created_at: string;
            updated_at: string;
        }],
        next_cursor: string;
        has_more: boolean;
        limit: number;

    }
}
